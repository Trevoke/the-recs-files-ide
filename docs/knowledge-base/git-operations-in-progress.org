#+TITLE: Git Operations Left in Progress — Reference
#+STARTUP: overview

Not a project document. This records which files git leaves under a worktree's
git dir while a merge, rebase, cherry-pick, revert or bisect is in progress,
and how to find them from a linked worktree, as established by a run of git
2.54.0 on 2026-10-15 and as git's documentation describes.

*What it establishes.* Each operation in progress leaves a marker under the git
dir of the worktree it runs in: =MERGE_HEAD= for a merge, =rebase-merge/= for a
rebase, =rebase-apply/= for a rebase with =--apply=, =CHERRY_PICK_HEAD= for a
cherry-pick (with =sequencer/= when it picks more than one commit),
=REVERT_HEAD= for a revert, and =BISECT_LOG= and =BISECT_START= for a bisect. A
linked worktree has a git dir of its own, under the common dir's
=worktrees/<id>/=, and its markers sit there; =git rev-parse --git-path <file>=
gives where a marker would be for the worktree it is run in.

* Each operation in progress leaves its own marker in the git dir

In the run below, each operation is left in progress in turn, and the markers
present are listed through =git rev-parse --git-path=:

| Operation                        | Markers                          |
|----------------------------------+----------------------------------|
| merge                            | =MERGE_HEAD=                     |
| rebase                           | =rebase-merge/=                  |
| rebase =--apply=                 | =rebase-apply/=                  |
| cherry-pick of one commit        | =CHERRY_PICK_HEAD=               |
| cherry-pick of several commits   | =CHERRY_PICK_HEAD=, =sequencer/= |
| revert                           | =REVERT_HEAD=                    |
| bisect                           | =BISECT_LOG=, =BISECT_START=     |

and none of them when nothing is in progress. The same holds in a linked
worktree.

* A linked worktree's markers are in its own git dir, found with --git-path

git-worktree(1), DETAILS:

#+begin_example
Within a linked worktree, $GIT_DIR is set to point to this private directory
(e.g. /path/main/.git/worktrees/test-next in the example) and $GIT_COMMON_DIR is
set to point back to the main worktree's $GIT_DIR (e.g. /path/main/.git).
...
Path resolution via git rev-parse --git-path uses either $GIT_DIR or
$GIT_COMMON_DIR depending on the path.
...
The rule of thumb is do not make any assumption about whether a path belongs to
$GIT_DIR or $GIT_COMMON_DIR when you need to directly access something inside
$GIT_DIR. Use git rev-parse --git-path to get the final path.
#+end_example

git-rev-parse(1), =--git-path <path>=: "Resolve "$GIT_DIR/<path>" and takes
other path relocation variables such as $GIT_OBJECT_DIRECTORY, $GIT_INDEX_FILE...
into account."

In the run below, a linked worktree's =--git-dir= is
=<common dir>/worktrees/linked= while its =--git-common-dir= is the main
worktree's =.git=; =--git-path MERGE_HEAD= there gives
=.git/worktrees/linked/MERGE_HEAD=; and a merge in
progress in the linked worktree leaves =MERGE_HEAD= in its own git dir, none in
the common dir, and no marker the main worktree sees.

* The run

Run with =bash ops-run.sh <empty directory>=; the script writes that directory
as =<tmp>= in its output.

#+begin_src sh
set -u
export GIT_CONFIG_GLOBAL=/dev/null GIT_CONFIG_NOSYSTEM=1 GIT_EDITOR=true
export GIT_AUTHOR_NAME=a GIT_AUTHOR_EMAIL=a@b GIT_COMMITTER_NAME=a GIT_COMMITTER_EMAIL=a@b
tmp=$(cd "$1" && pwd -P)
markers="MERGE_HEAD rebase-merge rebase-apply CHERRY_PICK_HEAD REVERT_HEAD sequencer BISECT_LOG BISECT_START"
present() {
  for m in $markers; do
    p=$(git rev-parse --path-format=absolute --git-path "$m")
    [ -e "$p" ] && printf ' %s' "$m"
  done
  echo
}
diverge() {
  git checkout --quiet -b other
  echo other > widget.el && git commit --quiet --all --message other
  echo more > other.el && git add other.el && git commit --quiet --message "other file"
  git checkout --quiet "$2"
  echo main > widget.el && git commit --quiet --all --message main
}
operation() {
  dir=$1; name=$2; shift 2
  ( cd "$dir"
    "$@" >/dev/null 2>&1
    printf '%-22s markers:' "$name"; present
    git merge --abort >/dev/null 2>&1; git rebase --abort >/dev/null 2>&1
    git cherry-pick --abort >/dev/null 2>&1; git revert --abort >/dev/null 2>&1
    git bisect reset >/dev/null 2>&1; git reset --quiet --hard >/dev/null 2>&1
    git checkout --quiet "$3" 2>/dev/null; true )
}
main() {
  cd "$tmp"
  git --version
  git init --quiet --initial-branch=main root && cd root
  echo base > widget.el && git add widget.el && git commit --quiet --message base
  ( diverge . main )
  git worktree add --quiet -b work ../linked main
  ( cd ../linked && git branch --quiet other-linked other )
  echo "== where the git dirs are"
  echo "main worktree:   --git-dir $(git rev-parse --path-format=absolute --git-dir), --git-common-dir $(git rev-parse --path-format=absolute --git-common-dir)"
  ( cd ../linked && echo "linked worktree: --git-dir $(git rev-parse --path-format=absolute --git-dir), --git-common-dir $(git rev-parse --path-format=absolute --git-common-dir)"
    echo "linked worktree: --git-path MERGE_HEAD $(git rev-parse --path-format=absolute --git-path MERGE_HEAD)" )
  echo "== each operation left in progress"
  for dir in . ../linked; do
    case $dir in .) label=main; branch=main; other=other;; *) label=linked; branch=work; other=other-linked;; esac
    echo "-- in the $label worktree"
    ( cd $dir; printf '%-22s markers:' "nothing"; present )
    operation $dir merge sh -c "git merge $other" $branch
    operation $dir "rebase" sh -c "git rebase $other" $branch
    operation $dir "rebase --apply" sh -c "git rebase --apply $other" $branch
    operation $dir cherry-pick sh -c "git cherry-pick $other~1" $branch
    operation $dir "cherry-pick sequence" sh -c "git cherry-pick $other~1 $other" $branch
    operation $dir revert sh -c "git revert --no-commit HEAD" $branch
    operation $dir bisect sh -c "git bisect start" $branch
  done
  echo "== the linked worktree's markers sit under its own git dir"
  ( cd ../linked && git merge other-linked >/dev/null 2>&1
    ls "$(git rev-parse --path-format=absolute --git-dir)" | grep -x MERGE_HEAD | sed 's/^/linked git dir holds: /'
    ls "$(git rev-parse --path-format=absolute --git-common-dir)" | grep -x MERGE_HEAD | sed 's/^/common git dir holds: /'
    echo "main worktree markers while the linked one merges:$(cd "$tmp/root" && present)"
    git merge --abort )
}
main 2>&1 | sed "s|$tmp|<tmp>|g"
#+end_src
